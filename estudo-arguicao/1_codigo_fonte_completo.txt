 36      public static String formatarReal(double v) {
 37          String s = String.format(java.util.Locale.ROOT, "%.2f", v);
 38          int ponto = s.indexOf('.');
 39          String inteiro = s.substring(0, ponto);
 40          String centavos = s.substring(ponto + 1);
 41  
 42          StringBuilder comPonto = new StringBuilder();
 43          int digitos = 0;
 44          for (int i = inteiro.length() - 1; i >= 0; i--) {
 45              comPonto.insert(0, inteiro.charAt(i));
 46              digitos++;
 47              if (digitos % 3 == 0 && i > 0) {
 48                  comPonto.insert(0, '.');
 49              }
 50          }
 51          return "R$ " + comPonto + "," + centavos;
 52      }

  1  package diarios.extratores;
  2  
  3  import diarios.modelo.Fato;
  4  import java.util.ArrayList;
  5  
  6  public abstract class Extrator {
  7  
  8      private final String tipo;
  9  
 10      protected Extrator(String tipo) {
 11          this.tipo = tipo;
 12      }
 13  
 14      public String getTipo() {
 15          return tipo;
 16      }
 17  
 18      public abstract ArrayList<Fato> extrair(String texto);
 19  
 20      protected Fato criar(String valor, int inicio, int fim) {
 21          return new Fato(tipo, valor, inicio, fim);
 22      }
 23  
 24      protected static boolean ehDigito(char c) {
 25          return c >= '0' && c <= '9';
 26      }
 27  
 28      protected static int contarDigitos(String texto, int posicao) {
 29          int n = 0;
 30          while (posicao + n < texto.length() && ehDigito(texto.charAt(posicao + n))) {
 31              n++;
 32          }
 33          return n;
 34      }
 35  
 36      protected static boolean casaMolde(String texto, int posicao, String molde) {
 37          if (posicao + molde.length() > texto.length()) {
 38              return false;
 39          }
 40          for (int i = 0; i < molde.length(); i++) {
 41              char esperado = molde.charAt(i);
 42              char achado = texto.charAt(posicao + i);
 43              if (esperado == '9' ? !ehDigito(achado) : achado != esperado) {
 44                  return false;
 45              }
 46          }
 47          return true;
 48      }
 49  
 50      public static String semAcento(String s) {
 51          return s.replace('ç', 'c').replace('ã', 'a').replace('á', 'a')
 52                  .replace('â', 'a').replace('à', 'a').replace('é', 'e')
 53                  .replace('ê', 'e').replace('í', 'i').replace('ó', 'o')
 54                  .replace('ô', 'o').replace('õ', 'o').replace('ú', 'u');
 55      }
 56  }

  1  package diarios.extratores;
  2  
  3  import diarios.modelo.Fato;
  4  import java.util.ArrayList;
  5  
  6  public class ExtratorData extends Extrator {
  7  
  8      private static final String[] MESES = {
  9          "janeiro", "fevereiro", "marco", "abril", "maio", "junho",
 10          "julho", "agosto", "setembro", "outubro", "novembro", "dezembro"
 11      };
 12  
 13      public ExtratorData() {
 14          super(Fato.DATA);
 15      }
 16  
 17      @Override
 18      public ArrayList<Fato> extrair(String texto) {
 19          ArrayList<Fato> achados = new ArrayList<>();
 20          acharNumericas(texto, achados);
 21          acharPorExtenso(texto, achados);
 22          return achados;
 23      }
 24  
 25      private void acharNumericas(String texto, ArrayList<Fato> achados) {
 26          for (int i = 0; i < texto.length(); i++) {
 27              if (casaMolde(texto, i, "99/99/9999")) {
 28                  achados.add(criar(texto.substring(i, i + 10), i, i + 10));
 29                  i += 9;
 30              } else if (casaMolde(texto, i, "99/99/99")) {
 31                  achados.add(criar(texto.substring(i, i + 8), i, i + 8));
 32                  i += 7;
 33              }
 34          }
 35      }
 36  
 37      private void acharPorExtenso(String texto, ArrayList<Fato> achados) {
 38          String base = semAcento(texto.toLowerCase());
 39          for (String mes : MESES) {
 40              int p = base.indexOf(mes);
 41              while (p >= 0) {
 42                  int inicio = recuarAteDia(base, p);
 43                  int fim = avancarAteAno(base, p + mes.length());
 44                  if (inicio >= 0 && fim > 0) {
 45                      achados.add(criar(texto.substring(inicio, fim), inicio, fim));
 46                  }
 47                  p = base.indexOf(mes, p + mes.length());
 48              }
 49          }
 50      }
 51  
 52      private int recuarAteDia(String texto, int posMes) {
 53          int p = posMes - 1;
 54          if (p < 4 || texto.charAt(p) != ' ') {
 55              return -1;
 56          }
 57          if (!texto.startsWith("de", p - 2)) {
 58              return -1;
 59          }
 60          p -= 3;
 61          if (p < 0 || texto.charAt(p) != ' ') {
 62              return -1;
 63          }
 64          int fimDia = p;
 65          while (p > 0 && (ehDigito(texto.charAt(p - 1)) || texto.charAt(p - 1) == 'º')) {
 66              p--;
 67          }

 14              if (casa(saida, i) && !digitoAntes(saida, i)

 16                  for (int k = 0; k < 11; k++) {
 17                      char c = saida.charAt(i + k);
 18                      if (c >= '0' && c <= '9') {
 19                          saida.setCharAt(i + k, '*');
 20                      }
 21                  }
 22                  i += MOLDE_CPF.length();
 23              } else {
 24                  i++;
 25              }
 26          }
 27          return saida.toString();
 28      }
 29  
 30      private static boolean casa(StringBuilder s, int p) {
 31          for (int k = 0; k < MOLDE_CPF.length(); k++) {
 32              char esperado = MOLDE_CPF.charAt(k);
 33              char achado = s.charAt(p + k);
 34              boolean ok = esperado == '9' ? (achado >= '0' && achado <= '9') : achado == esperado;
 35              if (!ok) {
 36                  return false;
 37              }
 38          }
 39          return true;
 40      }
 41  
 42      private static boolean digitoAntes(StringBuilder s, int p) {
 43          return p > 0 && Character.isDigit(s.charAt(p - 1));
 44      }
 45  
 46      private static boolean digitoDepois(StringBuilder s, int p) {
 47          return p < s.length() && Character.isDigit(s.charAt(p));
 48      }
 49  }

 22              Files.createDirectories(pasta);
 23          }
 24          Path destino = pasta.resolve(nomeArquivo);

 27                  escritor.write(l);
 28                  escritor.newLine();
 29              }
 30          }
 31          return destino.toString();

 32          "fica alterad", "ficam alterad", "passa a vigorar", "passam a vigorar",

 36      public static String de(Ato ato) {
 37          String texto = Extrator.semAcento(ato.getTexto().toLowerCase());
 38          String especie = ato.getEspecie();
 39  
 40          boolean temDinheiro = !ato.getFatosDoTipo(Fato.VALOR).isEmpty();
 41          boolean temEmpresa = !ato.getFatosDoTipo(Fato.CNPJ).isEmpty();
 42  
 43          if (temEmpresa && temDinheiro) {
 44              return CONTRATACAO;
 45          }
 46          if (contem(texto, MARCAS_ORCAMENTO)) {
 47              return ORCAMENTO;
 48          }
 49          if (contem(texto, MARCAS_PESSOAL)) {
 50              return PESSOAL;
 51          }
 52          if (contem(texto, MARCAS_LICITACAO) || especie.equals("EDITAL")) {
 53              return LICITACAO;
 54          }
 55          if (contem(texto, MARCAS_NORMA)
 56                  && (especie.equals("DECRETO") || especie.equals("LEI")
 57                      || especie.equals("LEI COMPLEMENTAR") || especie.equals("PORTARIA"))) {
 58              return NORMA;
 59          }
 60          return OUTRO;
 61      }

 63      private static boolean contem(String texto, String[] marcas) {
 64          for (String m : marcas) {
 65              if (texto.contains(m)) {
 66                  return true;
 67              }
 68          }
 69          return false;
 70      }

 95                  menorDistancia = d;
 96                  melhor = v;
 97              }
 98          }

112                  ultimaMarca = p;
113                  papel = par[1];
114              }
115          }
116          if (ultimaMarca < 0) {
117              return new String[] {INDEFINIDO, nomeAntesDe(texto, posicaoCnpj)};

123          int i = marca;
124          while (i < janela.length() && janela.charAt(i) != ':' && janela.charAt(i) != ' ') {
125              i++;
126          }
127          while (i < janela.length() && (janela.charAt(i) == ':' || janela.charAt(i) == ' ')) {
128              i++;
129          }
130          String resto = janela.substring(Math.min(i, janela.length()));
131  
132          int corte = resto.length();
133          String[] paradas = {",", ";", " sito", " SITO", " inscrit", " INSCRIT",
134                              "CNPJ", "cnpj", " – ", " - "};
135          for (String parada : paradas) {
136              int p = resto.indexOf(parada);
137              if (p > 0 && p < corte) {
138                  corte = p;
139              }
140          }
141          String nome = limparBordas(resto.substring(0, corte));
142          return nome.length() >= NOME_MINIMO && nome.length() <= NOME_MAXIMO
143                  ? nome : "(nao identificada)";

146      private String limparBordas(String s) {
147          String nome = s.trim();
148          while (!nome.isEmpty() && !Character.isLetterOrDigit(nome.charAt(0))) {
149              nome = nome.substring(1).trim();
150          }
151          while (!nome.isEmpty() && ":,;-–".indexOf(nome.charAt(nome.length() - 1)) >= 0) {
152              nome = nome.substring(0, nome.length() - 1).trim();
153          }
154          return tirarRestoDePalavra(nome);
155      }

157      private String tirarRestoDePalavra(String nome) {
158          int espaco = nome.indexOf(' ');
159          if (espaco < 0 || espaco > 3) {
160              return nome;
161          }
162          String primeira = nome.substring(0, espaco);
163          boolean itemDeLista = primeira.endsWith(")") || primeira.endsWith(":")
164                  || primeira.endsWith(".");
165          boolean pedacoDePalavra = primeira.length() <= 2
166                  && primeira.equals(primeira.toLowerCase());
167          return (itemDeLista || pedacoDePalavra) ? nome.substring(espaco + 1).trim() : nome;
168      }

173          String maiusculo = antes.toUpperCase();
174

176          for (String sufixo : SUFIXOS) {

178              int fim = p + sufixo.length();
179              if (p >= 0 && fim > fimDoNome && terminaAqui(maiusculo, fim)) {
180                  fimDoNome = fim;
181              }
182          }
183          if (fimDoNome > 0) {

185              if (nome.length() >= NOME_MINIMO) {

187              }

195

198              corte = Math.max(corte, ateMarca.lastIndexOf(sep));
199          }

201          while (!nome.isEmpty() && ",- ".indexOf(nome.charAt(nome.length() - 1)) >= 0) {
202              nome = nome.substring(0, nome.length() - 1).trim();
203          }

215          int p = fimDoNome;
216          int limite = Math.max(0, fimDoNome - NOME_MAXIMO);
217          int inicio = fimDoNome;
218          while (p > limite) {
219              char c = texto.charAt(p - 1);
220              boolean parteDoNome = Character.isLetterOrDigit(c) || " &-./".indexOf(c) >= 0;
221              if (!parteDoNome) {
222                  break;
223              }
224              p--;
225              if (c != ' ') {
226                  inicio = p;
227              }
228          }
229          String nome = texto.substring(inicio, fimDoNome).trim();
230          for (String ruido : RUIDOS) {
231              if (nome.toLowerCase().startsWith(ruido)) {

233              }
234          }
235          return nome;

246                      maiusculas++;

248              }

251      }

 42      private void montarResumo(ArrayList<Diario> diarios, PainelContratos painel,
 43                                GrafoNormas grafo) {
 44          int atos = 0;
 45          String periodo = periodoDe(diarios);
 46          ArrayList<String> municipios = new ArrayList<>();
 47  
 48          for (Diario d : diarios) {
 49              if (!d.isUtilizavel()) {
 50                  continue;
 51              }
 52              atos += d.getAtos().size();
 53              if (!municipios.contains(d.getMunicipio())) {
 54                  municipios.add(d.getMunicipio());
 55              }
 56          }
 57          if (atos == 0) {
 58              resumo = "Nenhum ato pode ser lido no periodo.";
 59              return;
 60          }
 61  
 62          int contratacoes = painel.getContratos().size();
 63          int atosDeContrato = contarAtosDoAssunto(diarios, Assunto.CONTRATACAO);
 64          int operacoes = grafo.doTipo(Relacao.ALTERA).size()
 65                  + grafo.doTipo(Relacao.REVOGA).size()
 66                  + grafo.doTipo(Relacao.PRORROGA).size();
 67          int normasAtingidas = grafo.maisAlteradas().size();
 68  
 69          StringBuilder s = new StringBuilder();
 70          s.append("Em ").append(periodo).append(", ");
 71          s.append(municipios.size() == 1 ? municipios.get(0) : municipios.size() + " municipios");
 72          s.append(" publicou ").append(atos).append(" atos. ");
 73  
 74          if (contratacoes > 0) {
 75              s.append("Em ").append(atosDeContrato)
 76               .append(atosDeContrato == 1 ? " deles ha " : " deles ha ")
 77               .append(contratacoes).append(contratacoes == 1 ? " contratacao" : " contratacoes")
 78               .append(", somando ").append(Contrato.formatarReal(painel.total())).append(". ");
 79          }
 80          if (operacoes > 0) {
 81              s.append(operacoes).append(operacoes == 1 ? " alteracao atinge " : " alteracoes atingem ")
 82               .append(normasAtingidas)
 83               .append(normasAtingidas == 1 ? " norma anterior. " : " normas anteriores. ");
 84          } else {
 85              s.append("Nenhuma norma anterior foi tocada. ");
 86          }
 87          resumo = s.toString().trim();
 88      }

 90      private void perfilDoDiario(ArrayList<Diario> diarios) {
 91          int total = 0;
 92          int contratacao = 0;
 93          int pessoal = 0;
 94          int norma = 0;
 95          int licitacao = 0;
 96          int orcamento = 0;
 97  
 98          for (Diario d : diarios) {
 99              if (!d.isUtilizavel()) {
100                  continue;
101              }
102              for (Ato a : d.getAtos()) {
103                  total++;
104                  String assunto = Assunto.de(a);
105                  if (assunto.equals(Assunto.CONTRATACAO)) {
106                      contratacao++;
107                  } else if (assunto.equals(Assunto.PESSOAL)) {
108                      pessoal++;
109                  } else if (assunto.equals(Assunto.NORMA)) {
110                      norma++;
111                  } else if (assunto.equals(Assunto.LICITACAO)) {
112                      licitacao++;
113                  } else if (assunto.equals(Assunto.ORCAMENTO)) {
114                      orcamento++;
115                  }
116              }
117          }
118          if (total == 0) {
119              return;
120          }
121  
122          int gastoPct = (contratacao + licitacao) * 100 / total;
123          int normaPct = norma * 100 / total;
124          String tipo = gastoPct > normaPct * 3
125                  ? "um diario de EXECUCAO, nao de legislacao"
126                  : "um diario com peso legislativo relevante";
127  
128          achados.add(new Achado(Achado.LEITURA,
129                  "O que este diario e",
130                  "E " + tipo + ": " + gastoPct + "% dos atos tratam de gastar dinheiro "
131                  + "(contrato ou licitacao) e " + normaPct + "% mexem em norma. "
132                  + "Quem acompanha a prefeitura por aqui esta vendo, sobretudo, "
133                  + "para onde o dinheiro vai.",
134                  "de " + total + " atos: " + contratacao + " contratacoes, " + licitacao
135                  + " licitacoes, " + pessoal + " pessoal, " + norma + " normas, "
136                  + orcamento + " orcamento"));
137      }

139      private void concentracaoDeGasto(PainelContratos painel) {
140          ArrayList<String[]> ranking = painel.ranking();
141          double total = painel.total();
142          if (ranking.size() < 3 || total <= 0) {
143              return;
144          }
145  
146          double top3 = 0;
147          StringBuilder nomes = new StringBuilder();
148          for (int i = 0; i < 3; i++) {
149              top3 += valorDe(ranking.get(i)[2]);
150              nomes.append(i > 0 ? ", " : "").append(ranking.get(i)[1]);
151          }
152          int pct = (int) Math.round(top3 * 100 / total);
153  
154          String leitura = pct >= CONCENTRACAO_ALTA
155                  ? "O gasto do periodo esta CONCENTRADO: tres empresas ficam com a "
156                    + "maior parte do valor, enquanto as outras " + (ranking.size() - 3)
157                    + " dividem o resto."
158                  : "O gasto do periodo esta DISTRIBUIDO entre os " + ranking.size()
159                    + " fornecedores: nenhum trio domina o valor.";
160  
161          achados.add(new Achado(pct >= CONCENTRACAO_ALTA ? Achado.ATENCAO : Achado.LEITURA,
162                  "Concentracao do gasto",
163                  leitura + " Os tres maiores somam " + pct + "% do total.",
164                  nomes + " = " + Contrato.formatarReal(top3) + " de "
165                  + Contrato.formatarReal(total)));

169          ArrayList<Contrato> contratos = painel.getContratos();
170          double total = painel.total();

300      private void oQueNaoFoiEntendido(ArrayList<Diario> diarios, Cobertura cobertura) {
301          if (cobertura.getTotalAtos() == 0) {
302              return;
303          }
304          int pct = cobertura.getDuvidosos() * 100 / cobertura.getTotalAtos();
305          if (pct < DUVIDA_ALTA) {

309          ArrayList<String> motivos = new ArrayList<>();
310          ArrayList<Integer> vezes = new ArrayList<>();
311          for (Diario d : diarios) {

313                  if (!a.getDecisao().equals(Ato.DUVIDOSO)) {
314                      continue;
315                  }
316                  String motivo = resumirMotivo(a.getMotivo());
317                  int i = motivos.indexOf(motivo);
318                  if (i < 0) {
319                      motivos.add(motivo);
320                      vezes.add(1);
321                  } else {
322                      vezes.set(i, vezes.get(i) + 1);
323                  }
324              }
325          }
326  
327          StringBuilder lista = new StringBuilder();
328          for (int i = 0; i < motivos.size(); i++) {
329              lista.append(lista.length() > 0 ? "; " : "")
330                   .append(vezes.get(i)).append("x ").append(motivos.get(i));
331          }
332  
333          achados.add(new Achado(Achado.LIMITE,
334                  "O que o programa ainda nao trata",
335                  pct + "% dos atos ficaram em duvida. Eles nao foram descartados nem "
336                  + "contados como certos: estao separados, com o motivo de cada um. "
337                  + "Esta e a lista do que falta melhorar, e ela so existe porque o "
338                  + "portao tem tres saidas em vez de duas.",
339                  lista.toString()));

362      private String periodoDe(ArrayList<Diario> diarios) {
363          String menor = null;
364          String maior = null;
365          for (Diario d : diarios) {
366              if (menor == null || d.getData().compareTo(menor) < 0) {
367                  menor = d.getData();
368              }
369              if (maior == null || d.getData().compareTo(maior) > 0) {
370                  maior = d.getData();
371              }
372          }
373          if (menor == null) {
374              return "periodo vazio";
375          }
376          return menor.equals(maior) ? menor : menor + " a " + maior;
377      }

379      private double valorDe(String formatado) {
380          StringBuilder n = new StringBuilder();
381          for (int i = 0; i < formatado.length(); i++) {
382              char c = formatado.charAt(i);
383              if (c >= '0' && c <= '9') {
384                  n.append(c);
385              } else if (c == ',') {
386                  n.append('.');
387              }
388          }
389          try {
390              return Double.parseDouble(n.toString());
391          } catch (NumberFormatException e) {
392              return 0.0;
393          }
394      }

140          String url = API + parametros;
141          for (int tentativa = 0; ; tentativa++) {
142              try {
143                  String resposta = buscarTexto(url, oQue);
144                  esperar();
145                  return resposta;
146              } catch (ColetaException e) {

148                  if (!e.podeTentarDeNovo() || acabaram) {
149                      throw new ColetaException(url, e.getMessage()
150                              + (tentativa > 0 ? " (depois de " + (tentativa + 1) + " tentativas)" : ""),
151                              e.podeTentarDeNovo(), e.getEsperaSugeridaMs());
152                  }
153                  long espera = Math.max(ESPERAS_ENTRE_TENTATIVAS_MS[tentativa], e.getEsperaSugeridaMs());
154                  esperar(Math.min(espera, ESPERA_MAXIMA_MS));
155              }

159      private String buscarTexto(String url, String oQue) throws ColetaException {
160          HttpRequest pedido = HttpRequest.newBuilder()
161                  .uri(URI.create(url))
162                  .header("User-Agent", AGENTE)
163                  .header("Accept", "application/json, text/plain, */*")
164                  .timeout(Duration.ofSeconds(90))
165                  .GET()
166                  .build();
167          try {
168              HttpResponse<String> resposta =
169                      cliente.send(pedido, HttpResponse.BodyHandlers.ofString(StandardCharsets.UTF_8));
170              int status = resposta.statusCode();
171              if (status != 200) {
172                  boolean passageiro = status == 429 || status >= 500;
173                  long sugerida = resposta.headers().firstValue("Retry-After")
174                          .map(this::segundosParaMs).orElse(0L);
175                  throw new ColetaException(url, oQue + " respondeu HTTP " + status
176                          + (status == 429 ? " (limite de pedidos da API)" : "")
177                          + " [" + trechoDoCorpo(resposta.body()) + "]", passageiro, sugerida);
178              }
179              return resposta.body();
180          } catch (IOException e) {
181              throw new ColetaException(url, oQue + " falhou: " + e.getClass().getSimpleName()
182                      + " " + e.getMessage(), true, 0);
183          } catch (InterruptedException e) {
184              Thread.currentThread().interrupt();
185              throw new ColetaException(url, oQue + " foi interrompido");
186          }
187      }

  5  public class Json {
  6  
  7      public static ArrayList<String> objetos(String resposta, String lista) {
  8          ArrayList<String> partes = new ArrayList<>();
  9          String marca = "\"" + lista + "\"";
 10          int p = resposta.indexOf(marca);
 11          if (p < 0) {
 12              return partes;
 13          }
 14          int abre = resposta.indexOf('[', p);
 15          if (abre < 0) {
 16              return partes;
 17          }
 18  
 19          int profundidade = 0;
 20          int inicio = -1;
 21          boolean dentroDeTexto = false;
 22  
 23          for (int i = abre; i < resposta.length(); i++) {
 24              char c = resposta.charAt(i);
 25  
 26              if (dentroDeTexto) {
 27                  if (c == '\\') {
 28                      i++;
 29                  } else if (c == '"') {
 30                      dentroDeTexto = false;
 31                  }
 32                  continue;
 33              }
 34              if (c == '"') {
 35                  dentroDeTexto = true;
 36              } else if (c == '{') {
 37                  if (profundidade == 0) {
 38                      inicio = i;
 39                  }
 40                  profundidade++;
 41              } else if (c == '}') {
 42                  profundidade--;
 43                  if (profundidade == 0 && inicio >= 0) {
 44                      partes.add(resposta.substring(inicio, i + 1));
 45                      inicio = -1;
 46                  }
 47              } else if (c == ']' && profundidade == 0) {
 48                  break;
 49              }
 50          }
 51          return partes;
 52      }
 53  
 54      public static String campo(String objeto, String chave) {
 55          String marca = "\"" + chave + "\"";
 56          int p = objeto.indexOf(marca);
 57          if (p < 0) {
 58              return "";
 59          }
 60          int i = objeto.indexOf(':', p + marca.length());
 61          if (i < 0) {
 62              return "";
 63          }
 64          i++;
 65          while (i < objeto.length() && objeto.charAt(i) == ' ') {
 66              i++;
 67          }
 68          if (i >= objeto.length()) {
 69              return "";
 70          }
 71          if (objeto.charAt(i) != '"') {
 72              return valorSemAspas(objeto, i);
 73          }
 74          i++;
 75  
 76          StringBuilder valor = new StringBuilder();
 77          while (i < objeto.length()) {
 78              char c = objeto.charAt(i);
 79              if (c == '\\' && i + 1 < objeto.length()) {
 80                  char seguinte = objeto.charAt(i + 1);
 81                  if (seguinte == 'n') {
 82                      valor.append('\n');
 83                  } else if (seguinte == 'u' && i + 5 < objeto.length()) {
 84                      valor.append((char) Integer.parseInt(objeto.substring(i + 2, i + 6), 16));
 85                      i += 4;
 86                  } else {
 87                      valor.append(seguinte);
 88                  }
 89                  i += 2;
 90                  continue;
 91              }
 92              if (c == '"') {
 93                  break;
 94              }
 95              valor.append(c);
 96              i++;
 97          }
 98          return valor.toString();
 99      }
100  
101      private static String valorSemAspas(String objeto, int inicio) {
102          int fim = inicio;
103          while (fim < objeto.length() && ",}]".indexOf(objeto.charAt(fim)) < 0) {
104              fim++;
105          }
106          String valor = objeto.substring(inicio, fim).trim();
107          return valor.equals("null") ? "" : valor;
108      }

  1  package diarios.servidor;
  2  
  3  import java.net.URLDecoder;
  4  import java.net.URLEncoder;
  5  import java.nio.charset.StandardCharsets;
  6  
  7  public class Formulario {
  8  
  9      public static String parametro(String dados, String chave) {
 10          if (dados == null || dados.isEmpty()) {
 11              return "";
 12          }
 13          for (String par : dados.split("&")) {
 14              int igual = par.indexOf('=');
 15              if (igual < 0) {
 16                  continue;
 17              }
 18              if (decodificar(par.substring(0, igual)).equals(chave)) {
 19                  return decodificar(par.substring(igual + 1)).trim();
 20              }
 21          }
 22          return "";
 23      }
 24  
 25      public static int inteiro(String dados, String chave, int padrao) {
 26          String valor = parametro(dados, chave);
 27          if (valor.isEmpty()) {
 28              return padrao;
 29          }
 30          try {
 31              return Integer.parseInt(valor);
 32          } catch (NumberFormatException e) {
 33              return padrao;
 34          }
 35      }
 36  
 37      public static String codificar(String s) {
 38          return URLEncoder.encode(s == null ? "" : s, StandardCharsets.UTF_8);
 39      }
 40  
 41      private static String decodificar(String s) {
 42          try {
 43              return URLDecoder.decode(s, StandardCharsets.UTF_8);
 44          } catch (IllegalArgumentException e) {
 45              return s;
 46          }
 47      }
 48  }

253                                     PainelContratos painel) {
254          ArrayList<String> municipios = new ArrayList<>();
255          for (Diario d : diarios) {

257                  municipios.add(d.getMunicipio());

262          h.add("<h2>Por municipio</h2>");
263          h.add("<p class=\"nota\">Os mesmos numeros do topo, separados por municipio. "

266                  + "<th class=\"num\">recusados</th><th class=\"num\">atos</th>"
267                  + "<th class=\"num\">aceitos</th><th class=\"num\">contratacoes</th>"
268                  + "<th class=\"num\">valor contratado</th></tr></thead><tbody>");
269          for (String m : municipios) {
270              int qtd = 0;
271              int recusados = 0;
272              int atos = 0;
273              int aceitos = 0;

277                  }

285              int contratos = 0;
286              double valor = 0;
287              for (Contrato c : painel.getContratos()) {

291                  }
292              }

342      static String classeDoStatus(String status) {
343          if (status.equals(EstadoFonte.COM_DADO)) {
344              return "ok";
345          }
346          if (status.equals(EstadoFonte.FALHOU)) {
347              return "rej";
348          }

350              return "duv";

353      }

355      static String bonito(String apelido) {
356          StringBuilder nome = new StringBuilder();
357          for (String palavra : apelido.split("-")) {
358              if (palavra.isEmpty()) {
359                  continue;
360              }
361              nome.append(nome.length() > 0 ? " " : "")
362                  .append(Character.toUpperCase(palavra.charAt(0))).append(palavra.substring(1));
363          }
364          return nome.toString();
365      }

139              }
140              String texto = d.getTextoLimpo();
141              String base = Extrator.semAcento(texto.toLowerCase());
142              int p = base.indexOf(alvo);
143              while (p >= 0 && achados < LIMITE_RESULTADOS) {
144                  achados++;
145                  int ato = atoNaPosicao(d, p);
146                  String trecho = Privacidade.mascararCpf(
147                          texto.substring(Math.max(0, p - 80), Math.min(texto.length(), p + alvo.length() + 80)))
148                          .replace('\n', ' ');
149                  String destino = ato >= 0 ? linkDoAto(d.getArquivo(), ato)
150                          : "/diario?arquivo=" + Formulario.codificar(d.getArquivo());
151                  linhas.add("<div class=\"fato\"><a href=\"" + destino + "\">" + esc(d.getArquivo())
152                          + (ato >= 0 ? " &middot; ato " + ato : " &middot; fora de ato") + "</a>"
153                          + "<div class=\"prova\">&hellip;" + esc(trecho) + "&hellip;</div></div>");

155              }

171      private String destacar(Ato ato) {
172          String texto = ato.getTexto();
173          ArrayList<int[]> trechos = new ArrayList<>();
174          for (Fato f : ato.getFatos()) {
175              trechos.add(new int[] {f.getInicio() - ato.getPosicao(), f.getFim() - ato.getPosicao()});
176          }
177          ordenar(trechos);
178  
179          StringBuilder s = new StringBuilder();
180          int cursor = 0;
181          for (int[] t : trechos) {
182              int ini = Math.max(0, Math.min(t[0], texto.length()));
183              int fim = Math.max(ini, Math.min(t[1], texto.length()));
184              if (ini < cursor) {
185                  continue;
186              }
187              s.append(esc(Privacidade.mascararCpf(texto.substring(cursor, ini))));
188              s.append("<mark>").append(esc(Privacidade.mascararCpf(texto.substring(ini, fim))))
189               .append("</mark>");
190              cursor = fim;
191          }
192          s.append(esc(Privacidade.mascararCpf(texto.substring(cursor))));
193          return s.toString();
194      }

196      private void ordenar(ArrayList<int[]> trechos) {
197          for (int i = 1; i < trechos.size(); i++) {
198              int[] atual = trechos.get(i);
199              int j = i - 1;
200              while (j >= 0 && trechos.get(j)[0] > atual[0]) {
201                  trechos.set(j + 1, trechos.get(j));
202                  j--;
203              }
204              trechos.set(j + 1, atual);
205          }
206      }

212                  return i;
213              }
214          }
215          return -1;
216      }

117      public Rodada coletar(int dias, boolean refazerTudo) {
118          int n = Math.max(1, Math.min(dias, DIAS_MAXIMO));
119          LocalDate hoje = LocalDate.now();
120          String desde = hoje.minusDays(n).toString();
121          String ate = hoje.toString();
122  
123          ArrayList<EstadoFonte> alvo = fontes.ligadas();
124          Rodada rodada = new Rodada(agora(), "coleta", n, desde, ate);
125  
126          int i = 0;
127          int falhasSeguidas = 0;
128          String interrompida = "";
129          for (EstadoFonte estado : alvo) {
130              i++;
131              if (!refazerTudo && estado.jaCobre(desde, ate)) {
132                  ResultadoColeta pulada = new ResultadoColeta(estado.getMunicipio());
133                  pulada.marcarPulada();
134                  pulada.anotar(estado.getMunicipio().getNome() + ": pulada, ja coletada ("
135                          + estado.getStatus() + " de " + estado.getCoberturaDesde() + " a "
136                          + estado.getCoberturaAte() + ")");
137                  rodada.adicionar(pulada);
138                  continue;
139              }
140              progresso = "coletando " + estado.getMunicipio().getNome()
141                      + " (" + i + " de " + alvo.size() + ")";
142              ResultadoColeta r = coletor.coletar(estado.getMunicipio(), desde, ate);
143              estado.registrarResultado(agora(), r, desde, ate);
144              rodada.adicionar(r);
145  
146              falhasSeguidas = r.isFalhouRede() ? falhasSeguidas + 1 : 0;
147              if (falhasSeguidas >= FALHAS_SEGUIDAS_PARA_PARAR && i < alvo.size()) {
148                  interrompida = "interrompida depois de " + FALHAS_SEGUIDAS_PARA_PARAR
149                          + " falhas seguidas; " + (alvo.size() - i)
150                          + " fonte(s) ficaram para a proxima rodada";
151                  break;
152              }
153          }
154          fontes.salvar();
155  
156          progresso = "lendo e interpretando os diarios da pasta";
157          processarPasta();
158  
159          String obs = alvo.isEmpty() ? "nenhuma fonte ligada" : interrompida;
160          rodada.finalizar(agora(), obs);
161          registrarRodada(rodada);
162          return rodada;
163      }

220      private ArrayList<String> quebrarTexto(String texto, int largura) {
221          ArrayList<String> linhas = new ArrayList<>();
222          StringBuilder atual = new StringBuilder();
223          for (String palavra : texto.split(" ")) {
224              if (atual.length() + palavra.length() + 1 > largura) {
225                  linhas.add(atual.toString());
226                  atual = new StringBuilder();
227              }
228              atual.append(atual.length() > 0 ? " " : "").append(palavra);
229          }
230          if (atual.length() > 0) {
231              linhas.add(atual.toString());
232          }
233          return linhas;
234      }

481          if (s == null) {
482              return "";
483          }

313      private static void testarPipelineCompleto() {
314          System.out.println("-- pipeline completo, sobre os diarios reais");
315  
316          ArrayList<String> falhas = new ArrayList<>();
317          ArrayList<Diario> lidos = new Ingestao(Aplicacao.PASTA_DIARIOS).lerTodos(falhas);
318          conferirMaiorQue("leu diarios da pasta", lidos.size(), 1);
319  
320          Limpeza limpeza = new Limpeza();
321          Extracao extracao = new Extracao();
322          Portao portao = new Portao();
323  
324          int recusados = 0;
325          ArrayList<Diario> ok = new ArrayList<>();
326  
327          for (Diario d : lidos) {
328              try {
329                  limpeza.limpar(d);
330              } catch (DiarioIlegivelException ex) {
331                  recusados++;
332                  continue;
333              }
334              new Fragmentacao().fragmentar(d);
335              extracao.extrair(d);
336              portao.avaliar(d);
337              ok.add(d);
338          }
339  
340          conferir("recusa o diario escaneado", 1, recusados);
341          conferirMaiorQue("processa diario bom", ok.size(), 0);
342  
343          Diario bh = ok.get(0);
344          conferirMaiorQue("encontrou atos", bh.getAtos().size(), 50);
345          conferirMaiorQue("descartou titulos sem corpo", bh.getTitulosDescartados(), 50);
346          conferirMaiorQue("extraiu fatos", contarFatos(bh), 500);
347          conferirMaiorQue("aceitou atos", bh.contar(Ato.ACEITO), 30);
348  
349          boolean todosDecididos = true;
350          for (Ato a : bh.getAtos()) {
351              if (a.getMotivo().isEmpty() || a.getDecisao().isEmpty()) {
352                  todosDecididos = false;
353              }
354          }
355          conferir("todo ato tem decisao e motivo", true, todosDecididos);
356  
357          ArrayList<String> identidades = new ArrayList<>();
358          boolean repetida = false;
359          for (Ato a : bh.getAtos()) {
360              if (identidades.contains(a.getIdentidade())) {
361                  repetida = true;
362              }
363              identidades.add(a.getIdentidade());
364          }
365          conferir("nenhuma identidade repetida no mesmo diario", false, repetida);
366  
367          boolean spansValidos = true;
368          String texto = bh.getTextoLimpo();
369          for (Ato a : bh.getAtos()) {
370              for (Fato f : a.getFatos()) {
371                  if (f.getInicio() < 0 || f.getFim() > texto.length()
372                          || f.getInicio() >= f.getFim()) {
373                      spansValidos = false;
374                  }
375              }
376          }
377          conferir("todo span cabe no texto", true, spansValidos);
378  
379          GrafoNormas grafo = new GrafoNormas();
380          grafo.processar(ok);
381          conferirMaiorQue("montou o grafo", grafo.getArestas().size(), 50);
382  
383          PainelContratos painel = new PainelContratos();
384          painel.processar(ok);
385          LeituraDoAto leitor = new LeituraDoAto(painel, grafo);
386          int comLeitura = 0;
387          int contratosLidos = 0;
388          for (Ato a : bh.getAtos()) {
389              String leitura = leitor.ler(a, bh.getData());
390              if (!leitura.isEmpty()) {
391                  comLeitura++;
392              }
393              if (leitura.startsWith("Contrata ")) {
394                  contratosLidos++;
395              }
396          }
397          conferir("todo ato ganha uma leitura", bh.getAtos().size(), comLeitura);
398          conferirMaiorQue("leitura reconhece contratacoes", contratosLidos, 5);
399          conferir("mesma entrada, mesma leitura (deterministico)",
400                  leitor.ler(bh.getAtos().get(0), bh.getData()),
401                  new LeituraDoAto(painel, grafo).ler(bh.getAtos().get(0), bh.getData()));
402      }

404      private static int contarFatos(Diario d) {
405          int n = 0;
406          for (Ato a : d.getAtos()) {
407              n += a.getFatos().size();
408          }
409          return n;
410      }

412      private static void conferir(String nome, Object esperado, Object obtido) {
413          marcar(esperado.equals(obtido), nome,
414                  "esperado <" + esperado + ">, obtido <" + obtido + ">");
415      }

417      private static void conferirMaiorQue(String nome, int obtido, int minimo) {
418          marcar(obtido > minimo, nome, "esperado > " + minimo + ", obtido " + obtido);
419      }

421      private static void marcar(boolean ok, String nome, String detalhe) {
422          if (ok) {
423              passou++;
424              System.out.println("   ok   " + nome);
425          } else {
426              falhou++;
427              System.out.println("   FALHOU  " + nome + " - " + detalhe);
428          }
